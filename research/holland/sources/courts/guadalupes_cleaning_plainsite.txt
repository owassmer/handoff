function psToggleDrawer() {
 var drawer = document.getElementById('psDrawer');
 var btn = document.getElementById('psHamburger');
 if (!drawer) return;
 var bar = document.querySelector('.psBar');
 if (bar) {
 var r = bar.getBoundingClientRect();
 drawer.style.top = r.bottom + 'px';
 drawer.style.maxHeight = (window.innerHeight - r.bottom - 10) + 'px';
 }
 var isOpen = drawer.classList.toggle('psDrawerOpen');
 if (btn) btn.setAttribute('aria-expanded', isOpen ? 'true' : 'false');
}
function psCloseDrawer() {
 var drawer = document.getElementById('psDrawer');
 var btn = document.getElementById('psHamburger');
 if (drawer) drawer.classList.remove('psDrawerOpen');
 if (btn) btn.setAttribute('aria-expanded', 'false');
}
document.addEventListener('click', function(e) {

 var btn = document.getElementById('psHamburger');
 if (drawer && drawer.classList.contains('psDrawerOpen')) {
 if (!drawer.contains(e.target) && btn && !btn.contains(e.target)) psCloseDrawer();
 }
});
window.addEventListener('resize', function() {
 if (window.innerWidth > 768) psCloseDrawer();
});
var psSearchHoverTimer = null;
var psSearchHoverSuppressed = false;
function psSearchHoverIn() {
 // Collapsing hides the box and puts the magnifying glass back exactly where
 // the pointer already is, so the browser fires mouseenter on a button the
 // user never moved onto and the box reopens the instant it is closed.
 // Hover stays disarmed until the pointer genuinely leaves the icon.
 if (psSearchHoverSuppressed) return;
 // Short delay on purpose. Expanding opens the box AND hides the nav links,
 // so firing on the first pixel of contact would rearrange the bar under a
 // pointer that was only crossing it on the way somewhere else. Resting on
 // the icon reads as intent; sweeping past does not.
 if (psSearchHoverTimer) clearTimeout(psSearchHoverTimer);
 psSearchHoverTimer = setTimeout(psExpandSearch, 150);
}
function psSearchHoverOut() {
 // The pointer really left, so a later entry is a deliberate one.
 psSearchHoverSuppressed = false;
 if (psSearchHoverTimer) { clearTimeout(psSearchHoverTimer); psSearchHoverTimer = null; }
}
function psExpandSearch() {
 psSearchHoverOut();
 var box = document.getElementById('psSearchBox');
 var btn = document.getElementById('psSearchToggle');
 var nav = document.getElementById('psBarNav');
 if (box) box.classList.add('psSearchOpen');
 if (btn) btn.style.display = 'none';
 if (nav) nav.classList.add('psBarNavHidden');
 var inp = document.getElementById('psSearchInput');
 if (inp) inp.focus();
 document.addEventListener('keydown', psSearchEscHandler);
}
function psCollapseSearch() {
 // The icon is about to reappear under the cursor -- disarm hover until the
 // pointer moves off it, or closing the box would immediately reopen it.
 psSearchHoverSuppressed = true;
 if (psSearchHoverTimer) { clearTimeout(psSearchHoverTimer); psSearchHoverTimer = null; }
 var box = document.getElementById('psSearchBox');
 var btn = document.getElementById('psSearchToggle');
 var nav = document.getElementById('psBarNav');
 if (box) box.classList.remove('psSearchOpen');
 if (btn) btn.style.display = '';
 if (nav) nav.classList.remove('psBarNavHidden');
 document.removeEventListener('keydown', psSearchEscHandler);
}
function psSearchEscHandler(e) {
 if (e.key === 'Escape') psCollapseSearch();
}
